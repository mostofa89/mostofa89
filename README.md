<summary><strong>🐍 Add the snake workflow: .github/workflows/snake.yml</strong></summary>

```yaml
name: Generate contribution snake

on:
  schedule:
    - cron: "0 */12 * * *"
  workflow_dispatch:
  push:
    branches:
      - main

permissions:
  contents: write

jobs:
  generate:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - name: Generate snake SVGs
        uses: Platane/snk/svg-only@v3
        with:
          github_user_name: mostofa89
          outputs: |
            dist/github-contribution-grid-snake.svg?color_snake=%238B5CF6&color_dots=%231a1b27%2C%230891b2%2C%236d28d9%2C%23a855f7%2C%23FF1493
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark&color_snake=%23FF1493

      - name: Publish SVGs to output branch
        uses: crazy-max/ghaction-github-pages@v4
        with:
          target_branch: output
          build_dir: dist
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
```
